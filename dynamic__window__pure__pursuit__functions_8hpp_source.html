65inline DynamicWindowBounds computeDynamicWindow(
66 const geometry_msgs::msg::Twist & current_speed,
67 const double & max_linear_vel,
68 const double & min_linear_vel,
69 const double & max_angular_vel,
70 const double & min_angular_vel,
71 const double & max_linear_accel,
72 const double & max_linear_decel,
73 const double & max_angular_accel,
74 const double & max_angular_decel,
75 const double & dt
76)
77{
78 DynamicWindowBounds dynamic_window;
79 constexpr double Eps = 1e-3;
80
81 // function to compute dynamic window for a single dimension
82 auto compute_window = [&](double vel, double max_vel, double min_vel, double accel, double decel)
83 {
84 double candidate_max_vel = 0.0;
85 double candidate_min_vel = 0.0;
86
87 if (vel > Eps) {
88 // if the current velocity is positive, acceleration means an increase in speed
89 candidate_max_vel = vel + accel * dt;
90 candidate_min_vel = vel - decel * dt;
91 } else if (vel < -Eps) {
92 // if the current velocity is negative, acceleration means a decrease in speed
93 candidate_max_vel = vel + decel * dt;
94 candidate_min_vel = vel - accel * dt;
95 } else {
96 // if the current velocity is zero, allow acceleration in both directions.
97 candidate_max_vel = vel + accel * dt;
98 candidate_min_vel = vel - accel * dt;
99 }
100
101 // clip to max/min velocity limits
102 double dynamic_window_max_vel = std::min(candidate_max_vel, max_vel);
103 double dynamic_window_min_vel = std::max(candidate_min_vel, min_vel);
104 return std::make_tuple(dynamic_window_max_vel, dynamic_window_min_vel);
105 };
106
107 // linear velocity
108 std::tie(dynamic_window.max_linear_vel, dynamic_window.min_linear_vel) =
109 compute_window(
110 current_speed.linear.x, max_linear_vel, min_linear_vel,
111 max_linear_accel, max_linear_decel);
112
113 // angular velocity
114 std::tie(dynamic_window.max_angular_vel, dynamic_window.min_angular_vel) =
115 compute_window(
116 current_speed.angular.z, max_angular_vel, min_angular_vel,
117 max_angular_accel, max_angular_decel);
118
119 return dynamic_window;
120}

127inline void applyRegulationToDynamicWindow(
128 const double & regulated_linear_vel,
129 DynamicWindowBounds & dynamic_window)
130{
131 // Create regulated bounds [0, v_reg] or [v_reg, 0]
132 double v_reg_min = std::min(0.0, regulated_linear_vel);
133 double v_reg_max = std::max(0.0, regulated_linear_vel);
134
135 // Intersect the dynamic window with the regulated bounds
136 dynamic_window.min_linear_vel = std::max(dynamic_window.min_linear_vel, v_reg_min);
137 dynamic_window.max_linear_vel = std::min(dynamic_window.max_linear_vel, v_reg_max);
138
139 // If min > max, collapse to the nearest boundary
140 if (dynamic_window.min_linear_vel > dynamic_window.max_linear_vel) {
141 if (dynamic_window.min_linear_vel > v_reg_max) {
142 dynamic_window.max_linear_vel = dynamic_window.min_linear_vel;
143 } else {
144 dynamic_window.min_linear_vel = dynamic_window.max_linear_vel;
145 }
146 }
147}

156inline std::tuple<double, double> computeOptimalVelocityWithinDynamicWindow(
157 const DynamicWindowBounds & dynamic_window,
158 const double & curvature,
159 const double & sign
160)
161{
162 double optimal_linear_vel;
163 double optimal_angular_vel;
164
165 // consider linear_vel - angular_vel space (horizontal and vertical axes respectively)
166 // Select the closest point to the line
167 // angular_vel = curvature * linear_vel within the dynamic window.
168 // If multiple points are equally close, select the one with the largest linear_vel.
169
170 // When curvature == 0, the line is angular_vel = 0
171 if (std::abs(curvature) < 1e-3) {
172 // linear velocity
173 if (sign >= 0.0) {
174 // If moving forward, select the max linear vel
175 optimal_linear_vel = dynamic_window.max_linear_vel;
176 } else {
177 // If moving backward, select the min linear vel
178 optimal_linear_vel = dynamic_window.min_linear_vel;
179 }
180
181 // angular velocity
182 // If the line angular_vel = 0 intersects the dynamic window,angular_vel = 0.0
183 if (dynamic_window.min_angular_vel <= 0.0 && 0.0 <= dynamic_window.max_angular_vel) {
184 optimal_angular_vel = 0.0;
185 } else {
186 // If not, select angular vel within dynamic window closest to 0
187 if (std::abs(dynamic_window.min_angular_vel) <= std::abs(dynamic_window.max_angular_vel)) {
188 optimal_angular_vel = dynamic_window.min_angular_vel;
189 } else {
190 optimal_angular_vel = dynamic_window.max_angular_vel;
191 }
192 }
193 return std::make_tuple(optimal_linear_vel, optimal_angular_vel);
194 }
195
196 // When the dynamic window and the line angular_vel = curvature * linear_vel intersect,
197 // select the intersection point that yields the highest linear velocity.
198
199 // List the four candidate intersection points
200 std::pair<double, double> candidates[] = {
201 {dynamic_window.min_linear_vel, curvature * dynamic_window.min_linear_vel},
202 {dynamic_window.max_linear_vel, curvature * dynamic_window.max_linear_vel},
203 {dynamic_window.min_angular_vel / curvature, dynamic_window.min_angular_vel},
204 {dynamic_window.max_angular_vel / curvature, dynamic_window.max_angular_vel}
205 };
206
207 double best_linear_vel = -std::numeric_limits<double>::max() * sign;
208 double best_angular_vel = 0.0;
209
210 for (auto [linear_vel, angular_vel] : candidates) {
211 // Check whether the candidate lies within the dynamic window
212 if (linear_vel >= dynamic_window.min_linear_vel &&
213 linear_vel <= dynamic_window.max_linear_vel &&
214 angular_vel >= dynamic_window.min_angular_vel &&
215 angular_vel <= dynamic_window.max_angular_vel)
216 {
217 // Select the candidate with the largest linear velocity (considering moving direction)
218 if (linear_vel * sign > best_linear_vel * sign) {
219 best_linear_vel = linear_vel;
220 best_angular_vel = angular_vel;
221 }
222 }
223 }
224
225 // If best_linear_vel was updated, it means that a valid intersection exists
226 if (best_linear_vel != -std::numeric_limits<double>::max() * sign) {
227 optimal_linear_vel = best_linear_vel;
228 optimal_angular_vel = best_angular_vel;
229 return std::make_tuple(optimal_linear_vel, optimal_angular_vel);
230 }
231
232 // When the dynamic window and the line angular_vel = curvature * linear_vel have no intersection,
233 // select the point within the dynamic window that is closest to the line.
234
235 // Because the dynamic window is a convex region,
236 // the closest point must be one of its four corners.
237 const std::array<std::array<double, 2>, 4> corners = {
238 {
239 {dynamic_window.min_linear_vel, dynamic_window.min_angular_vel},
240 {dynamic_window.min_linear_vel, dynamic_window.max_angular_vel},
241 {dynamic_window.max_linear_vel, dynamic_window.min_angular_vel},
242 {dynamic_window.max_linear_vel, dynamic_window.max_angular_vel}
243 }};
244
245 // Compute the distance from a point (linear_vel, angular_vel)
246 // to the line angular_vel = curvature * linear_vel
247 const double denom = std::sqrt(curvature * curvature + 1.0);
248 auto compute_dist = [&](const std::array<double, 2> & corner) -> double {
249 return std::abs(curvature * corner[0] - corner[1]) / denom;
250 };
251
252 double closest_dist = std::numeric_limits<double>::max();
253 best_linear_vel = -std::numeric_limits<double>::max() * sign;
254 best_angular_vel = 0.0;
255
256 for (const auto & corner : corners) {
257 const double dist = compute_dist(corner);
258 // Update if this corner is closer to the line,
259 // or equally close but has a larger linear velocity (considering moving direction)
260 if (dist < closest_dist ||
261 (std::abs(dist - closest_dist) <= 1e-3 && corner[0] * sign > best_linear_vel * sign))
262 {
263 closest_dist = dist;
264 best_linear_vel = corner[0];
265 best_angular_vel = corner[1];
266 }
267 }
268
269 optimal_linear_vel = best_linear_vel;
270 optimal_angular_vel = best_angular_vel;
271
272 return std::make_tuple(optimal_linear_vel, optimal_angular_vel);
273}

292inline std::tuple<double, double> computeDynamicWindowVelocities(
293 const geometry_msgs::msg::Twist & current_speed,
294 const double & max_linear_vel,
295 const double & min_linear_vel,
296 const double & max_angular_vel,
297 const double & min_angular_vel,
298 const double & max_linear_accel,
299 const double & max_linear_decel,
300 const double & max_angular_accel,
301 const double & max_angular_decel,
302 const double & regulated_linear_vel,
303 const double & curvature,
304 const double & sign,
305 const double & dt
306)
307{
308 // compute Dynamic Window
309 DynamicWindowBounds dynamic_window = computeDynamicWindow(
310 current_speed, max_linear_vel, min_linear_vel, max_angular_vel, min_angular_vel,
311 max_linear_accel, max_linear_decel, max_angular_accel, max_angular_decel, dt);
312
313 // apply regulation to Dynamic Window
314 applyRegulationToDynamicWindow(regulated_linear_vel, dynamic_window);
315
316 // compute optimal velocity within Dynamic Window
317 auto [linear_vel, angular_vel] = computeOptimalVelocityWithinDynamicWindow(
318 dynamic_window, curvature, sign);
319
320 return std::make_tuple(linear_vel, angular_vel);
321}
