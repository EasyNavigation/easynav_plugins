50inline double calculateCurvature(double lookahead_x, double lookahead_y)
51{
52 const double carrot_dist2 = lookahead_x * lookahead_x + lookahead_y * lookahead_y;
53 if (carrot_dist2 > 0.001) {
54 return 2.0 * lookahead_y / carrot_dist2;
55 } else {
56 return 0.0;
57 }
58}

68inline double curvatureConstraint(
69 const double raw_linear_vel, const double curvature, const double min_radius)
70{
71 const double radius = std::fabs(1.0 / curvature);
72 if (radius < min_radius) {
73 return raw_linear_vel * (1.0 - (std::fabs(radius - min_radius) / min_radius));
74 } else {
75 return raw_linear_vel;
76 }
77}

89inline double obstacleConstraint(
90 const double raw_linear_vel,
91 const double min_obstacle_distance,
92 const double obstacle_scaling_dist,
93 const double obstacle_scaling_gain)
94{
95 if (!std::isfinite(min_obstacle_distance)) {
96 return raw_linear_vel;
97 }
98
99 if (min_obstacle_distance < obstacle_scaling_dist) {
100 const double clamped_dist = std::max(0.0, min_obstacle_distance);
101 return raw_linear_vel * (obstacle_scaling_gain * clamped_dist / obstacle_scaling_dist);
102 }
103
104 return raw_linear_vel;
105}

112inline double approachVelocityScalingFactor(
113 const double remaining_path_distance,
114 const double euclidean_dist_to_goal,
115 const double approach_velocity_scaling_dist)
116{
117 // Waiting to apply the threshold based on integrated distance ensures we don't erroneously
118 // apply approach scaling on curvy paths still far from the goal in a straight line.
119 if (remaining_path_distance < approach_velocity_scaling_dist) {
120 return euclidean_dist_to_goal / approach_velocity_scaling_dist;
121 } else {
122 return 1.0;
123 }
124}

133inline double approachVelocityConstraint(
134 const double constrained_linear_vel,
135 const double remaining_path_distance,
136 const double euclidean_dist_to_goal,
137 const double min_approach_velocity,
138 const double approach_velocity_scaling_dist)
139{
140 const double velocity_scaling = approachVelocityScalingFactor(
141 remaining_path_distance, euclidean_dist_to_goal, approach_velocity_scaling_dist);
142 double approach_vel = constrained_linear_vel * velocity_scaling;
143
144 if (approach_vel < min_approach_velocity) {
145 approach_vel = min_approach_velocity;
146 }
147
148 return std::min(constrained_linear_vel, approach_vel);
149}
