84inline nav_msgs::msg::Path polyline(const std::vector<Eigen::Vector3d> & pts, double res = 0.05)
85{
86 nav_msgs::msg::Path path;
87 path.header.frame_id = easynav::RTTFBuffer::getInstance()->get_tf_info().map_frame;
88 for (std::size_t i = 1; i < pts.size(); ++i) {
89 const int n = std::max(1, static_cast<int>((pts[i] - pts[i - 1]).norm() / res));
90 for (int k = (i == 1 ? 0 : 1); k <= n; ++k) {
91 geometry_msgs::msg::PoseStamped p;
92 p.header.frame_id = path.header.frame_id;
93 const Eigen::Vector3d q = pts[i - 1] + (pts[i] - pts[i - 1]) * k / n;
94 p.pose.position.x = q.x();
95 p.pose.position.y = q.y();
96 p.pose.position.z = q.z();
97 p.pose.orientation.w = 1.0;
98 path.poses.push_back(p);
99 }
100 }
101 return path;
102}

104inline double distance_to(const Eigen::Vector3d & p, const std::vector<Eigen::Vector3d> & pts)
105{
106 double best = std::numeric_limits<double>::infinity();
107 for (std::size_t i = 1; i < pts.size(); ++i) {
108 const Eigen::Vector3d e = pts[i] - pts[i - 1];
109 const double t = std::clamp(
110 (p - pts[i - 1]).dot(e) / std::max(e.squaredNorm(), 1e-12), 0.0, 1.0);
111 best = std::min(best, (p - (pts[i - 1] + t * e)).norm());
112 }
113 return best;
114}

116inline Result run(const Scenario & sc)
117{
118 static int instance = 0;
119 const bool aerial = sc.model == "aerial";
120 const bool ackermann = sc.model == "ackermann";
121 easynav::RobotGeometryRegistry::getInstance()->set_geometry(sc.geometry);
122
123 std::vector<rclcpp::Parameter> params = sc.params;
124 params.emplace_back("mpc.motion_model", sc.model);
125 auto node = std::make_shared<easynav::ControllerNode>(
126 rclcpp::NodeOptions().arguments(
127 {"--ros-args", "-r", "__node:=mpc_sim_" + std::to_string(instance++)})
128 .parameter_overrides(params));
129 easynav::MPCController mpc;
130 mpc.initialize(node, "mpc");
131 const auto limits = node->get_robot_limits();
132
133 const auto & tf_info = easynav::RTTFBuffer::getInstance()->get_tf_info();
134 easynav::NavState nav_state;
135 std::vector<Eigen::Vector3d> path_pts = sc.path;
136 if (!aerial) {
137 for (auto & p : path_pts) {
138 p.z() = 0.0;
139 }
140 }
141 nav_state.set("path", polyline(path_pts));
142 nav_state.set("goal_tolerance.position", sc.goal_tolerance);
143 nav_state.set("goal_tolerance.yaw", 2.0 * M_PI);
144
145 const Eigen::Vector3d goal = path_pts.back();
146 Eigen::Vector3d pos = sc.start;
147 if (!aerial) {pos.z() = 0.0;}
148 double yaw = sc.yaw;
149 Eigen::Vector4d vel = Eigen::Vector4d::Zero(); // vx, vy, vz, wz (body)
150 geometry_msgs::msg::Twist last_cmd;
151
152 const double dt = 1.0 / 30.0;
153 Result r;
154 double dev_sum = 0.0;
155 int n = 0;
156 for (double t = 0.0; t < sc.timeout; t += dt) {
157 nav_msgs::msg::Odometry odom;
158 odom.header.frame_id = tf_info.map_frame;
159 odom.pose.pose.position.x = pos.x();
160 odom.pose.pose.position.y = pos.y();
161 odom.pose.pose.position.z = pos.z();
162 odom.pose.pose.orientation.z = std::sin(yaw / 2.0);
163 odom.pose.pose.orientation.w = std::cos(yaw / 2.0);
164 nav_state.set("robot_pose", odom);
165
166 geometry_msgs::msg::TransformStamped tf;
167 tf.header.frame_id = tf_info.map_frame;
168 tf.child_frame_id = tf_info.robot_frame;
169 tf.transform.translation.x = pos.x();
170 tf.transform.translation.y = pos.y();
171 tf.transform.translation.z = pos.z();
172 tf.transform.rotation = odom.pose.pose.orientation;
173 easynav::RTTFBuffer::getInstance()->setTransform(tf, "mpc_sim", true);
174 if (!sc.obstacles.empty() || !sc.body_points.empty()) {
175 easynav::PointPerception perception;
176 perception.frame_id = tf_info.map_frame;
177 perception.stamp = node->now();
178 perception.valid = true;
179 for (const auto & o : sc.obstacles) {
180 perception.data.push_back(pcl::PointXYZ(o.x(), o.y(), o.z()));
181 }
182 for (const auto & b : sc.body_points) {
183 const Eigen::Vector3d w = pos + Eigen::Vector3d(
184 std::cos(yaw) * b.x() - std::sin(yaw) * b.y(),
185 std::sin(yaw) * b.x() + std::cos(yaw) * b.y(), b.z());
186 perception.data.push_back(pcl::PointXYZ(w.x(), w.y(), w.z()));
187 }
188 nav_state.set("points", perception);
189 }
190
191 mpc.update_rt(nav_state);
192 const auto cmd = nav_state.get<geometry_msgs::msg::TwistStamped>("cmd_vel").twist;
193 r.max_solve_ms = std::max(r.max_solve_ms, mpc.last_result().solve_ms);
194 const auto & qp = mpc.last_result();
195 if (qp.variables > 0 && // it ran (not turning in place, or at the goal)
196 (qp.status == easynav::mpc::QPStatus::Infeasible ||
197 qp.status == easynav::mpc::QPStatus::Failed))
198 {
199 ++r.qp_failures;
200 }
201 r.max_linear_jump = std::max(r.max_linear_jump, std::abs(cmd.linear.x - last_cmd.linear.x));
202 if (std::abs(cmd.linear.x) + std::abs(cmd.linear.y) + std::abs(cmd.linear.z) < 0.02) {
203 r.max_rotation_in_place = std::max(r.max_rotation_in_place, std::abs(cmd.angular.z));
204 }
205 if (limits.min_turning_radius > 0.0 && std::abs(cmd.linear.x) > 1e-3) {
206 r.max_curvature_excess = std::max(
207 r.max_curvature_excess,
208 std::abs(cmd.angular.z) - std::abs(cmd.linear.x) / limits.min_turning_radius);
209 }
210 last_cmd = cmd;
211 if ((sc.verbose || std::getenv("MPC_SIM_VERBOSE")) && static_cast<int>(t / dt) % 5 == 0) {
212 std::printf(
213 "t=%5.2f pos=(%6.2f,%6.2f,%5.2f) yaw=%6.2f cmd=(%5.2f,%5.2f,%5.2f,%5.2f) st=%d\n", t,
214 pos.x(), pos.y(), pos.z(), yaw, cmd.linear.x, cmd.linear.y, cmd.linear.z, cmd.angular.z,
215 static_cast<int>(mpc.last_result().status));
216 }
217
218 // The robot: its velocity follows the command (with the lag), within the robot limits
219 Eigen::Vector4d target(
220 std::clamp(cmd.linear.x, limits.min_linear_vel, limits.max_linear_vel),
221 std::clamp(cmd.linear.y, -limits.max_lateral_vel, limits.max_lateral_vel),
222 std::clamp(cmd.linear.z, -limits.max_vertical_vel, limits.max_vertical_vel),
223 std::clamp(cmd.angular.z, -limits.max_angular_vel, limits.max_angular_vel));
224 if (sc.model == "diff" || ackermann) {target(1) = 0.0;}
225 if (!aerial) {target(2) = 0.0;}
226 const double a = sc.lag > 0.0 ? std::exp(-dt / sc.lag) : 0.0;
227 vel = a * vel + (1.0 - a) * target;
228 if (ackermann) {
229 // Steering at the rear axle: w = v tan(delta) / L, |delta| within the turning radius
230 const double max_delta = limits.min_turning_radius > 0.0 ?
231 std::atan(sc.wheelbase / limits.min_turning_radius) : M_PI / 2.0 - 1e-3;
232 const double delta = std::abs(vel(0)) > 1e-6 ?
233 std::clamp(std::atan(sc.wheelbase * vel(3) / vel(0)), -max_delta, max_delta) : 0.0;
234 vel(3) = vel(0) * std::tan(delta) / sc.wheelbase;
235 }
236 const int sub = 10;
237 for (int i = 0; i < sub; ++i) {
238 const double h = dt / sub;
239 const double c = std::cos(yaw), s = std::sin(yaw);
240 pos.x() += (vel(0) * c - vel(1) * s) * h;
241 pos.y() += (vel(0) * s + vel(1) * c) * h;
242 pos.z() += vel(2) * h;
243 yaw += vel(3) * h;
244 }
245
246 const double dev = distance_to(pos, path_pts);
247 dev_sum += dev;
248 ++n;
249 r.dev_max = std::max(r.dev_max, dev);
250 r.v_max = std::max(r.v_max, std::hypot(vel(0), vel(1)));
251 for (const auto & o : sc.obstacles) {
252 const Eigen::Vector3d d = o - pos;
253 // Clearance from the robot's circle (ground: in the plane)
254 const double dist = aerial ? d.norm() : d.head<2>().norm();
255 r.min_clearance = std::min(r.min_clearance, dist - sc.geometry.radius);
256 }
257 r.time = t;
258 if ((goal - pos).norm() < sc.goal_tolerance &&
259 std::abs(vel(0)) + std::abs(vel(1)) + std::abs(vel(2)) < 0.05)
260 {
261 r.reached = true;
262 break;
263 }
264 }
265 r.dev_mean = n ? dev_sum / n : 0.0;
266 r.final_position = pos;
267 r.final_yaw = yaw;
268 easynav::RobotGeometryRegistry::getInstance()->set_geometry(easynav::RobotGeometry{});
269 return r;
270}
