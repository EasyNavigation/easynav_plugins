47inline geometry_msgs::msg::PoseStamped pose_stamped(double x, double y, double yaw)
48{
49 geometry_msgs::msg::PoseStamped p;
50 p.header.frame_id = "map";
51 p.pose.position.x = x;
52 p.pose.position.y = y;
53 tf2::Quaternion q;
54 q.setRPY(0.0, 0.0, yaw);
55 p.pose.orientation.x = q.x();
56 p.pose.orientation.y = q.y();
57 p.pose.orientation.z = q.z();
58 p.pose.orientation.w = q.w();
59 return p;
60}

63inline nav_msgs::msg::Path polyline(
64 const std::vector<std::pair<double, double>> & corners, double goal_yaw)
65{
66 nav_msgs::msg::Path path;
67 path.header.frame_id = "map";
68 for (size_t i = 0; i + 1 < corners.size(); ++i) {
69 const auto [ax, ay] = corners[i];
70 const auto [bx, by] = corners[i + 1];
71 const double len = std::hypot(bx - ax, by - ay);
72 const double yaw = std::atan2(by - ay, bx - ax);
73 const int n = std::max(1, static_cast<int>(std::ceil(len / 0.05)));
74 for (int k = 0; k < n; ++k) {
75 const double f = static_cast<double>(k) / n;
76 path.poses.push_back(pose_stamped(ax + f * (bx - ax), ay + f * (by - ay), yaw));
77 }
78 }
79 const auto [gx, gy] = corners.back();
80 path.poses.push_back(pose_stamped(gx, gy, goal_yaw));
81 return path;
82}

85inline double distance_to(
86 const std::vector<std::pair<double, double>> & corners, double x, double y)
87{
88 double best = std::numeric_limits<double>::infinity();
89 for (size_t i = 0; i + 1 < corners.size(); ++i) {
90 const auto [ax, ay] = corners[i];
91 const auto [bx, by] = corners[i + 1];
92 const double dx = bx - ax;
93 const double dy = by - ay;
94 const double len2 = dx * dx + dy * dy;
95 const double f = len2 > 0.0 ? std::clamp(((x - ax) * dx + (y - ay) * dy) / len2, 0.0, 1.0) :
96 0.0;
97 best = std::min(best, std::hypot(x - (ax + f * dx), y - (ay + f * dy)));
98 }
99 return best;
100}

103inline void add_disc(pcl::PointCloud<pcl::PointXYZ> & cloud, double cx, double cy, double r)
104{
105 for (int i = 0; i < 24; ++i) {
106 const double a = 2.0 * M_PI * i / 24;
107 cloud.push_back(pcl::PointXYZ(cx + r * std::cos(a), cy + r * std::sin(a), 0.2));
108 }
109 cloud.push_back(pcl::PointXYZ(cx, cy, 0.2));
110}

113inline void add_wall(
114 pcl::PointCloud<pcl::PointXYZ> & cloud, double ax, double ay, double bx, double by)
115{
116 const int n = std::max(1, static_cast<int>(std::hypot(bx - ax, by - ay) / 0.05));
117 for (int i = 0; i <= n; ++i) {
118 const double f = static_cast<double>(i) / n;
119 cloud.push_back(pcl::PointXYZ(ax + f * (bx - ax), ay + f * (by - ay), 0.2));
120 }

124inline void set_rectangle(MPPIParams & p, double length, double width)
125{
126 const double hx = length / 2.0;
127 const double hy = width / 2.0;
128 p.footprint = {{hx, hy}, {-hx, hy}, {-hx, -hy}, {hx, -hy}};
129 p.robot_radius = std::hypot(hx, hy);
130 p.inscribed_radius = std::min(hx, hy);
131 p.safety_radius = p.robot_radius + 0.1;

172struct Sim
173{
174 easynav::MPPIOptimizer optimizer;
175 MPPIParams params;
176 Pose2D pose;
177 Velocity velocity;
178 Velocity last_cmd;
179 nav_msgs::msg::Path path;
180 pcl::PointCloud<pcl::PointXYZ> obstacles;
181 HeadingMode heading_mode {HeadingMode::Path};
182 double heading_yaw {0.0};
183 double xy_tolerance {0.1};
184 double yaw_tolerance {0.1};
185 double period {1.0 / 30.0};
186 double time {0.0};
187
188 // What happened.
189 std::vector<Pose2D> trace;
190 std::vector<Velocity> cmds;
191 std::vector<MPPIStatus> statuses;
192 double min_clearance {std::numeric_limits<double>::infinity()};
193 bool limits_ok {true};
194 bool acceleration_ok {true};
195
196 explicit Sim(const MPPIParams & p, Pose2D start = {})
197 : params(p), pose(start)
198 {
199 optimizer.configure(params);
200 }
201
202 static double step_axis(double v, double target, double acc, double decel, double dt)
203 {
204 const bool slowing = target * v < 0.0 || std::abs(target) < std::abs(v);
205 const double max_delta = (slowing ? decel : acc) * dt;
206 double next = v + std::clamp(target - v, -max_delta, max_delta);
207 if ((v > 0.0 && next < 0.0) || (v < 0.0 && next > 0.0)) {
208 next = 0.0;
209 }
210 return next;
211 }
212
213 MPPIStatus step()
214 {
215 easynav::MPPIInput in;
216 in.pose = pose;
217 in.velocity = last_cmd;
218 in.path = &path;
219 in.obstacles = &obstacles;
220 in.heading_mode = heading_mode;
221 in.heading_yaw = heading_yaw;
222 in.elapsed = time > 0.0 ? period : 0.0;
223 in.xy_goal_tolerance = xy_tolerance;
224 in.yaw_goal_tolerance = yaw_tolerance;
225 const auto r = optimizer.compute(in);
226 const auto & p = optimizer.params();
227 const auto & c = r.cmd;
228 constexpr double eps = 1e-9;
229 if (c.vx > p.max_vx + eps || c.vx < p.min_vx - eps || std::abs(c.vy) > p.max_vy + eps ||
230 std::abs(c.wz) > p.max_wz + eps)
231 {
232 limits_ok = false;
233 }
234 // Holonomic: no direction faster than the configured ones.
235 if (p.motion_model == MotionModel::Omni) {
236 const double lx = c.vx >= 0.0 ? p.max_vx : -p.min_vx;
237 if (lx > 0.0 && p.max_vy > 0.0 && std::hypot(c.vx / lx, c.vy / p.max_vy) > 1.0 + 1e-6) {
238 limits_ok = false;
239 }
240 }
241 if (r.status == MPPIStatus::Following) {
242 // Omni: a step may be projected back into the ellipse of the speed limits, a bit more.
243 const double projection = p.motion_model == MotionModel::Omni ? 2e-3 : 0.0;
244 const double dv = std::max(p.acc_vx, p.decel_vx) * p.dt + eps + projection;
245 const double dy = std::max(p.acc_vy, p.decel_vy) * p.dt + eps + projection;
246 const double dw = std::max(p.acc_wz, p.decel_wz) * p.dt + eps;
247 if (std::abs(c.vx - last_cmd.vx) > dv || std::abs(c.vy - last_cmd.vy) > dy ||
248 std::abs(c.wz - last_cmd.wz) > dw)
249 {
250 acceleration_ok = false;
251 }
252 }
253 last_cmd = c;
254 cmds.push_back(c);
255 statuses.push_back(r.status);
256
257 // The robot: smoothed velocity, integrated.
258 velocity.vx = step_axis(velocity.vx, c.vx, p.acc_vx, p.decel_vx, period);
259 velocity.vy = step_axis(velocity.vy, c.vy, p.acc_vy, p.decel_vy, period);
260 velocity.wz = step_axis(velocity.wz, c.wz, p.acc_wz, p.decel_wz, period);
261 const double yaw = pose.yaw + 0.5 * velocity.wz * period;
262 pose.x += (velocity.vx * std::cos(yaw) - velocity.vy * std::sin(yaw)) * period;
263 pose.y += (velocity.vx * std::sin(yaw) + velocity.vy * std::cos(yaw)) * period;
264 pose.yaw = normalize(pose.yaw + velocity.wz * period);
265 time += period;
266 trace.push_back(pose);
267
268 // Clearance of the robot's body (its footprint, or its circle).
269 const auto & fp = optimizer.params().footprint;
270 const double cos_yaw = std::cos(pose.yaw);
271 const double sin_yaw = std::sin(pose.yaw);
272 for (const auto & o : obstacles) {
273 double clearance;
274 if (fp.empty()) {
275 clearance = std::hypot(o.x - pose.x, o.y - pose.y) - optimizer.params().robot_radius;
276 } else {
277 const double lx = cos_yaw * (o.x - pose.x) + sin_yaw * (o.y - pose.y);
278 const double ly = -sin_yaw * (o.x - pose.x) + cos_yaw * (o.y - pose.y);
279 const double border = easynav::footprint_border_distance(fp, lx, ly);
280 clearance = easynav::footprint_contains(fp, lx, ly) ? -border : border;
281 }
282 min_clearance = std::min(min_clearance, clearance);
283 }
284 return r.status;
285 }
286
288 bool run(double max_time)
289 {
290 while (time < max_time) {
291 if (step() == MPPIStatus::Arrived) {
292 return true;
293 }
294 }
295 return false;
296 }
297
298 double max_abs_vy() const
299 {
300 double m = 0.0;
301 for (const auto & c : cmds) {m = std::max(m, std::abs(c.vy));}
302 return m;
303 }
304};
